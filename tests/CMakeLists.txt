cmake_minimum_required(VERSION 3.16)
project(emote_physics_tests LANGUAGES CXX)
enable_testing()
add_executable(emote_physics_test physics_test.cpp)
target_compile_features(emote_physics_test PRIVATE cxx_std_17)
target_include_directories(emote_physics_test PRIVATE ..)
# These assertions remain enabled for a Release test build.
if(MSVC)
    target_compile_options(emote_physics_test PRIVATE /UNDEBUG)
else()
    target_compile_options(emote_physics_test PRIVATE -UNDEBUG)
endif()
add_test(NAME emote_physics COMMAND emote_physics_test)

add_executable(emote_frame_selection_test frame_selection_test.cpp)
target_compile_features(emote_frame_selection_test PRIVATE cxx_std_17)
target_include_directories(emote_frame_selection_test PRIVATE ..)
if(MSVC)
    target_compile_options(emote_frame_selection_test PRIVATE /UNDEBUG)
else()
    target_compile_options(emote_frame_selection_test PRIVATE -UNDEBUG)
endif()
add_test(NAME emote_frame_selection COMMAND emote_frame_selection_test)
